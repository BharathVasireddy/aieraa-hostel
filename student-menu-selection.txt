        // Initialize app
        document.addEventListener('DOMContentLoaded', function() {
            generateDayRibbon();
            updateCountdown();
            setupEventListeners();
            
            // Update countdown every minute
            setInterval(updateCountdown, 60000);

        function generateDayRibbon() {
            const ribbon = document.getElementById('day-ribbon');
            const today = new Date();
            
            // Generate 7 days (3 past, today, 3 future)
            for (let i = -3; i <= 3; i++) {
                const date = new Date(today);
                date.setDate(today.getDate() + i);
                
                const dayChip = document.createElement('div');
                dayChip.className = 'day-chip';
                dayChip.dataset.date = date.toISOString().split('T')[0];
                
                if (i < 0) {
                    dayChip.classList.add('past');
                } else if (i === 0) {
                    dayChip.classList.add('today');
                } else {
                    dayChip.classList.add('future');
                }
                
                // Tomorrow is selected by default
                if (i === 1) {
                    dayChip.classList.add('selected');
                }
                
                const dayName = date.toLocaleDateString('en', { weekday: 'short' });
                const dayNumber = date.getDate();
                
                dayChip.innerHTML = `
                    <div class="day-name">${dayName}</div>
                    <div class="day-number">${dayNumber}</div>
                `;
                
                // Add status indicators for past days
                if (i < 0) {
                    const status = Math.random() > 0.3 ? 'complete' : 'missed';
                    dayChip.innerHTML += `<div class="day-status status-${status}">
                        ${status === 'complete' ? '✓' : '✗'}
                    </div>`;
                }
                
                // Only add click listener for today and future dates
                if (i >= 0) {
                    dayChip.addEventListener('click', () => selectDay(dayChip, date));
                } else {
                    dayChip.style.cursor = 'default';
                    dayChip.style.opacity = '0.7';
                }
                
                ribbon.appendChild(dayChip);
            }
            
            // Scroll to show tomorrow in center (instead of today)
            setTimeout(() => {
                const tomorrowChip = ribbon.querySelector('.day-chip.selected');
                if (tomorrowChip) {
                    tomorrowChip.scrollIntoView({ inline: 'center', behavior: 'smooth' });
                }
            }, 100);
        }

        function selectDay(chipElement, date) {
            const today = new Date();
            today.setHours(0, 0, 0, 0);
            const selectedDate = new Date(date);
            selectedDate.setHours(0, 0, 0, 0);
            
            // Prevent selection of past dates
            if (selectedDate < today) {
                return;
            }
            
            // Remove previous selection
            document.querySelectorAll('.day-chip.selected').forEach(chip => {
                chip.classList.remove('selected');
            });
            
            // Add selection to clicked day
            chipElement.classList.add('selected');
            selectedDay = date;
            
            // Update date display
            updateSelectedDate();
            
            // Reset meal selections for new day
            mealSelections = {};
            updateMealCards();
        }

        function updateSelectedDate() {
            const title = document.getElementById('selected-date-title');
            const subtitle = document.getElementById('selected-date-subtitle');
            
            const today = new Date();
            today.setHours(0, 0, 0, 0);
            const tomorrow = new Date(today);
            tomorrow.setDate(today.getDate() + 1);
            const selectedDate = new Date(selectedDay);
            selectedDate.setHours(0, 0, 0, 0);
            
            if (selectedDate.getTime() === tomorrow.getTime()) {
                title.textContent = "Tomorrow, " + selectedDay.toLocaleDateString('en', { 
                    month: 'long', 
                    day: 'numeric' 
                });
                subtitle.textContent = "Choose your meals before 5:00 PM";
            } else if (selectedDate.getTime() === today.getTime()) {
                title.textContent = "Today, " + selectedDay.toLocaleDateString('en', { 
                    month: 'long', 
                    day: 'numeric' 
                });
                subtitle.textContent = "Selections are locked";
            } else if (selectedDate < today) {
                title.textContent = selectedDay.toLocaleDateString('en', { 
                    weekday: 'long',
                    month: 'long', 
                    day: 'numeric' 
                });
                subtitle.textContent = "Past day - view your meal history";
            } else {
                title.textContent = selectedDay.toLocaleDateString('en', { 
                    weekday: 'long',
                    month: 'long', 
                    day: 'numeric' 
                });
                subtitle.textContent = "Select your meals in advance";
            }
        }

        function updateMealCards() {
            ['breakfast', 'lunch', 'dinner'].forEach(meal => {
                const card = document.querySelector(`[data-meal="${meal}"]`);
                const selection = document.getElementById(`${meal}-selection`);
                
                if (mealSelections[meal]) {
                    card.classList.add('selected');
                    selection.textContent = mealSelections[meal].name;
                    selection.classList.remove('empty');
                } else {
                    card.classList.remove('selected');
                    selection.textContent = "Tap to choose your menu";
                    selection.classList.add('empty');
                }
            });
        }

        function updateCountdown() {
            const now = new Date();
            const today5PM = new Date(now);
            today5PM.setHours(17, 0, 0, 0);
            
            const progressBar = document.getElementById('countdown-progress');
            const countdownText = document.getElementById('countdown-text');
            
            if (now > today5PM) {
                // After 5 PM - show tomorrow's deadline
                const tomorrow5PM = new Date(today5PM);
                tomorrow5PM.setDate(tomorrow5PM.getDate() + 1);
                
                const timeLeft = tomorrow5PM - now;
                const hoursLeft = Math.floor(timeLeft / (1000 * 60 * 60));
                const minutesLeft = Math.floor((timeLeft % (1000 * 60 * 60)) / (1000 * 60));
                
                countdownText.textContent = `${hoursLeft}h ${minutesLeft}m left to lock tomorrow's meals`;
                
                // Progress based on 24-hour cycle
                const progress = ((24 * 60 * 60 * 1000) - timeLeft) / (24 * 60 * 60 * 1000) * 100;
                progressBar.style.width = Math.min(progress, 100) + '%';
            } else {
                // Before 5 PM today
                const timeLeft = today5PM - now;
                const hoursLeft = Math.floor(timeLeft / (1000 * 60 * 60));
                const minutesLeft = Math.floor((timeLeft % (1000 * 60 * 60)) / (1000 * 60));
                
                countdownText.textContent = `${hoursLeft}h ${minutesLeft}m left to lock tomorrow's meals`;
                
                // Progress from 9 AM to 5 PM (8 hours)
                const startTime = new Date(now);
                startTime.setHours(9, 0, 0, 0);
                const elapsed = now - startTime;
                const total = today5PM - startTime;
                const progress = (elapsed / total) * 100;
                
                progressBar.style.width = Math.max(0, Math.min(progress, 100)) + '%';
                
                // Add pulsing animation if close to deadline
                if (hoursLeft === 0 && minutesLeft <= 30) {
                    progressBar.style.animation = 'pulse 0.5s ease-in-out infinite alternate';
                }
            }
        }

        function setupEventListeners() {
            // Meal card clicks
            document.querySelectorAll('.meal-card').forEach(card => {
                card.addEventListener('click', () => {
                    const meal = card.dataset.meal;
                    openCategoryPanel(meal);
                });
            });

            // Overlay click to close panel
            document.getElementById('overlay').addEventListener('click', closeCategoryPanel);

            // Confirm button
            document.getElementById('confirm-btn').addEventListener('click', confirmSelection);

            // Panel handle for closing
            document.querySelector('.panel-handle').addEventListener('click', closeCategoryPanel);
        }

        function openCategoryPanel(meal) {
            currentMeal = meal;
            const panel = document.getElementById('category-panel');
            const overlay = document.getElementById('overlay');
            const title = document.getElementById('panel-title');
            const container = document.getElementById('categories-container');
            
            title.textContent = `Choose ${meal.charAt(0).toUpperCase() + meal.slice(1)} Menu`;
            
            // Clear previous categories
            container.innerHTML = '';
            
            // Populate categories for this meal
            const categories = menuData[meal] || [];
            categories.forEach(category => {
                const categoryCard = createCategoryCard(category);
                container.appendChild(categoryCard);
            });
            
            // Show panel
            overlay.classList.add('active');
            panel.classList.add('active');
            
            // Reset selection
            selectedCategory = null;
            updateConfirmButton();
        }

        function createCategoryCard(category) {
            const card = document.createElement('div');
            card.className = 'category-card';
            card.dataset.categoryId = category.id;
            
            card.innerHTML = `
                <div class="category-header">
                    <div class="category-info">
                        <h3>${category.name}</h3>
                        <p>${category.description}</p>
                    </div>
                    <div class="category-image">${category.emoji}</div>
                </div>
                <div class="menu-items">
                    <div class="items-grid">
                        ${category.items.map(item => `
                            <div class="menu-item">
                                <div class="item-emoji">${item.emoji}</div>
                                <div>${item.name}</div>
                            </div>
                        `).join('')}
                    </div>
                </div>
            `;
            
            card.addEventListener('click', () => {
                // Remove previous selection
                document.querySelectorAll('.category-card.selected').forEach(c => {
                    c.classList.remove('selected');
                });
                
                // Select this category
                card.classList.add('selected');
                selectedCategory = category;
                updateConfirmButton();
            });
            
            // Long press to show ingredients (mobile)
            let pressTimer;
            card.addEventListener('touchstart', (e) => {
                pressTimer = setTimeout(() => {
                    showIngredients(category);
                }, 800);
            });
            
            card.addEventListener('touchend', () => {
                clearTimeout(pressTimer);
            });
            
            return card;
        }

        function showIngredients(category) {
            // Vibrate if available
            if (navigator.vibrate) {
                navigator.vibrate(50);
            }
            
            alert(`${category.name} includes:\n${category.items.map(item => item.name).join(', ')}\n\nTap to select this menu option.`);
        }

        function updateConfirmButton() {
            const confirmBtn = document.getElementById('confirm-btn');
            confirmBtn.disabled = !selectedCategory;
            
            if (selectedCategory) {
                confirmBtn.textContent = `Confirm ${selectedCategory.name}`;
            } else {
                confirmBtn.textContent = 'Confirm Selection';
            }
        }

        function confirmSelection() {
            if (!selectedCategory) return;
            
            // Add loading state
            const confirmBtn = document.getElementById('confirm-btn');
            confirmBtn.classList.add('loading');
            confirmBtn.textContent = 'Confirming...';
            
            // Simulate API call
            setTimeout(() => {
                // Save selection
                mealSelections[currentMeal] = selectedCategory;
                
                // Update UI
                updateMealCards();
                closeCategoryPanel();
                showSuccessPopup();
                
                // Remove loading state
                confirmBtn.classList.remove('loading');
                updateConfirmButton();
            }, 1000);
        }

        function closeCategoryPanel() {
            const panel = document.getElementById('category-panel');
            const overlay = document.getElementById('overlay');
            
            panel.classList.remove('active');
            overlay.classList.remove('active');
            
            selectedCategory = null;
        }

        function showSuccessPopup() {
            const popup = document.getElementById('success-popup');
            const message = document.getElementById('success-message');
            
            message.textContent = `Your ${currentMeal} is locked in!`;
            popup.classList.add('active');
            
            // Auto hide after 2 seconds
            setTimeout(() => {
                popup.classList.remove('active');
            }, 2000);
            
            // Update streak if all meals selected
            checkStreakProgress();
        }

        function checkStreakProgress() {
            const totalMeals = Object.keys(mealSelections).length;
            if (totalMeals === 3) {
                // All meals selected for the day
                setTimeout(() => {
                    updateStreakDisplay();
                }, 2500);
            }
        }

        function updateStreakDisplay() {
            const streakCount = document.getElementById('streak-count');
            // Simulate streak increment
            streakCount.textContent = '8 Day Streak! 🎉';
            
            // Add celebration animation
            const flame = document.querySelector('.streak-flame');
            flame.style.animation = 'bounce 0.6s ease';
            
            setTimeout(() => {
                flame.style.animation = 'flicker 2s ease-in-out infinite alternate';
            }, 600);
        }

        // Touch gestures for category cards (swipe to see more photos)
        let startX, startY;
        
        document.addEventListener('touchstart', function(e) {
            startX = e.touches[0].clientX;
            startY = e.touches[0].clientY;
        });
        
        document.addEventListener('touchmove', function(e) {
            if (!startX || !startY) return;
            
            const diffX = startX - e.touches[0].clientX;
            const diffY = startY - e.touches[0].clientY;
            
            // If horizontal swipe on category card
            if (Math.abs(diffX) > Math.abs(diffY) && Math.abs(diffX) > 50) {
                const target = e.target.closest('.category-card');
                if (target) {
                    // Visual feedback for swipe
                    target.style.transform = `translateX(${-diffX * 0.1}px)`;
                    target.style.transition = 'none';
                }
            }
        });
        
        document.addEventListener('touchend', function(e) {
            const target = e.target.closest('.category-card');
            if (target) {
                target.style.transform = '';
                target.style.transition = '';
            }
            startX = null;
            startY = null;
        });
